║ 6️⃣ Wildcards in Generics                         ║
║   ?  = Unknown type                              ║
║   ? extends Type → Upper bound (read only)       ║
║   ? super Type   → Lower bound (write allowed)   ║
║                                                   
║   Examples:                                       ║
║   List<?> l1;               // any type           ║

║   List<? super Integer> l3; // Integer & Parents  ║

// Generic Class with Type T
class Box<T> {
    private T value;

    // Generic Constructor
    public Box(T value) {
        this.value = value;
    }

    public void setValue(T value) {
        this.value = value;
    }

    public T getValue() {
        return value;
    }

    // Normal Generic Method
    public <U> void showPair(U secondValue) {
        System.out.println("Box Value: " + value + ", Second Value: " + secondValue);
    }

    // STATIC GENERIC METHOD
    public static <X> void display(X item) {
        System.out.println("Static Generic Method prints: " + item);
    }
}

// A class to demonstrate Wildcards
class PrintService {
    
    // Method using Wildcard: unknown type (?)
    public void printAnyBox(Box<?> box) {
        System.out.println("Wildcard Box Value: " + box.getValue());
    }

    // Method using Upper Bound Wildcard (? extends Number)
    // Accepts Integer, Float, Double etc.
    public void printNumber(Box<? extends Number> box) {
        Number n = box.getValue();
        System.out.println("Number from NumberBox: " + n);
    }
}

public class Main {
    public static void main(String[] args) {

        // Creating Boxes
        Box<Integer> intBox = new Box<>(100);
        Box<String> strBox = new Box<>("Hello Java Generics");
        Box<Double> doubleBox = new Box<>(45.78);

        // Using normal generic method
        intBox.showPair("Score");
        strBox.showPair(2025);

        // Using Static Generic Method
        Box.display("Static Call");
        Box.display(12345);

        // Wildcard Usage
        PrintService ps = new PrintService();
        ps.printAnyBox(strBox);   // any type works
        ps.printAnyBox(intBox);

        // Upper bound wildcard -> only Numbers allowed
        ps.printNumber(intBox);
        ps.printNumber(doubleBox);

        // Compile Error ❌ if try:
        // ps.printNumber(strBox); // Not allowed because String is not a Number
    }
}
